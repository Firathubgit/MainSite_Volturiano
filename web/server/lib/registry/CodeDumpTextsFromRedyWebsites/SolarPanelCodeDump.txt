
# Codebase Structure and Rendering Order

This document contains the exact code for the application, ordered by the rendering hierarchy.

## 1. Entry Point
- **File**: `index.tsx`
- **Description**: The application entry point where React mounts the `App` component to the DOM.

## 2. Main Application Component
- **File**: `App.tsx`
- **Description**: The root component that defines the overall layout and renders child components in a specific order.

## 3. Shared Types
- **File**: `types.ts`
- **Description**: TypeScript definitions used across the application.

## 4. Components in Rendering Order
The `App` component renders the following components in this sequence:
1. `VideoBackground` (Background layer)
2. `Navbar` (Fixed navigation overlay)
3. `Hero` (Top section)
4. `Process` (Process explanation section)
5. `Reviews` (Testimonials and reviews section)
6. `ScrollReveal` (Scrollytelling philosophy section)
7. `WorkShowcase` (Projects and case studies)
8. `Footer` (Footer and contact info)

## 5. UI Components
- `Reveal` (Used by multiple components for scroll animations)

## File: src/components/ScrollReveal.tsx
```tsx
import React, { useRef, useEffect, useState } from 'react';

interface ScrollRevealProps {
  text: string;
}

export const ScrollReveal: React.FC<ScrollRevealProps> = ({ text }) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    const handleScroll = () => {
      if (!containerRef.current) return;

      const { top, height } = containerRef.current.getBoundingClientRect();
      const windowHeight = window.innerHeight;
      
      // Calculate how far the element has scrolled relative to the viewport
      const scrollDistance = -top;
      const totalScrollableDistance = height - windowHeight;

      let currentProgress = scrollDistance / totalScrollableDistance;

      // Clamp value between 0 and 1
      currentProgress = Math.max(0, Math.min(1, currentProgress));
      
      setProgress(currentProgress);
    };

    window.addEventListener('scroll', handleScroll);
    handleScroll(); // Initial check

    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  // Split text into words, then characters for granular control
  const words = text.split(" ");

  // Calculate total characters to map progress
  const totalChars = text.replace(/\s/g, "").length;
  let charCount = 0;

  return (
    <div ref={containerRef} className="relative h-[300vh] bg-black">
      
      {/* Sticky Container */}
      <div className="sticky top-0 h-screen flex flex-col items-center justify-center overflow-hidden">
        
        {/* Dynamic Background Orb */}
        <div 
            className="absolute inset-0 pointer-events-none z-0 flex items-center justify-center"
        >
            <div 
                className="w-[100vw] h-[100vw] rounded-full"
                style={{
                    // Using the consistent purple palette
                    background: 'radial-gradient(circle, #7c3aed 0%, #2e1065 40%, #000000 80%)', 
                    opacity: progress, 
                    transform: `scale(${Math.max(0, progress * 4)})`, 
                    transition: 'transform 0.1s linear, opacity 0.1s linear',
                    filter: 'blur(80px)'
                }}
            />
        </div>

        {/* BOTTOM CONNECTOR GRADIENT */}
        {/* Made aggressive: Starts transparent, becomes solid #2e1065 at 85% to ensure the last 15% is pure color */}
        <div 
            className="absolute bottom-0 left-0 right-0 h-[60vh] z-0 pointer-events-none"
            style={{
                background: 'linear-gradient(to bottom, transparent 0%, #2e1065 85%, #2e1065 100%)',
                opacity: Math.min(1, progress * 1.5) // Reaches full opacity faster
            }}
        />

        {/* Text Container */}
        <div className="relative z-10 max-w-6xl px-6 md:px-12 text-center mix-blend-screen">
            {/* Tagline */}
            <div className="mb-16 overflow-hidden">
                <span className="inline-block text-xs font-mono text-white/70 uppercase tracking-[0.3em] transition-all duration-700 transform"
                      style={{ opacity: progress > 0.05 ? 1 : 0, transform: `translateY(${progress > 0.05 ? '0' : '20px'})` }}>
                    The Philosophy
                </span>
            </div>

            {/* Main Reveal Text */}
            <h2 className="text-4xl md:text-6xl lg:text-7xl font-medium leading-[1.1] tracking-tight flex flex-wrap justify-center gap-x-[0.25em] gap-y-[0.1em]">
            {words.map((word, wordIndex) => {
                return (
                <span key={wordIndex} className="inline-flex whitespace-nowrap">
                    {word.split("").map((char, charIndex) => {
                    // Determine the threshold for this specific character
                    const currentGlobalCharIndex = charCount;
                    charCount++;
                    
                    // Finish revealing earlier to enjoy the full background state
                    const revealStart = (currentGlobalCharIndex / totalChars) * 0.8;
                    const revealEnd = revealStart + 0.1; 

                    // Calculate state based on progress
                    let opacity = 0; 
                    let blur = 12;
                    let y = 20;

                    if (progress > revealEnd) {
                        opacity = 1;
                        blur = 0;
                        y = 0;
                    } else if (progress > revealStart) {
                        const localProgress = (progress - revealStart) / (revealEnd - revealStart);
                        opacity = localProgress;
                        blur = 12 * (1 - localProgress);
                        y = 20 * (1 - localProgress);
                    }

                    return (
                        <span
                        key={charIndex}
                        style={{ 
                            color: 'white', 
                            opacity: opacity,
                            filter: `blur(${blur}px)`,
                            transform: `translateY(${y}px)`
                        }}
                        className="transition-all duration-100 ease-out inline-block"
                        >
                        {char}
                        </span>
                    );
                    })}
                </span>
                );
            })}
            </h2>
        </div>
      </div>
    </div>
  );
};
```

```tsx
import React, { useEffect, useRef, useState } from 'react';

interface RevealProps {
  children: React.ReactNode;
  width?: "fit-content" | "100%";
  delay?: number;
  className?: string;
}

export const Reveal: React.FC<RevealProps> = ({ 
  children, 
  width = "fit-content", 
  delay = 0,
  className = ""
}) => {
  const [isVisible, setIsVisible] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.unobserve(entry.target);
        }
      },
      {
        threshold: 0.1,
        rootMargin: "0px 0px -50px 0px" 
      }
    );

    if (ref.current) {
      observer.observe(ref.current);
    }

    return () => {
      if (ref.current) {
        observer.unobserve(ref.current);
      }
    };
  }, []);

  return (
    <div 
      ref={ref} 
      style={{ width, transitionDelay: `${delay}ms` }} 
      className={`${className} transition-all duration-1000 ease-out transform ${
        isVisible ? "opacity-100 translate-y-0 blur-0" : "opacity-0 translate-y-12 blur-sm"
      }`}
    >
      {children}
    </div>
  );
};
```
