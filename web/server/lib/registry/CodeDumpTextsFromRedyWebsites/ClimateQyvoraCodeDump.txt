### /components/ScrollText.tsx
import React, { useEffect, useRef } from 'react';

export const ScrollText: React.FC = () => {
  const containerRef = useRef<HTMLDivElement>(null);
  const textTrackRef = useRef<HTMLDivElement>(null);
  const letterRefs = useRef<(HTMLSpanElement | null)[]>([]);

  const words = ["CLIMATE", "AWARENESS", "ACTION", "PLATFORM"];

  useEffect(() => {
    let animationFrameId: number;
    let currentProgress = 0;
    let targetProgress = 0;

    const animate = () => {
      if (!containerRef.current || !textTrackRef.current) return;

      const containerRect = containerRef.current.getBoundingClientRect();
      const windowHeight = window.innerHeight;
      const windowWidth = window.innerWidth;

      // Calculate target progress based on scroll position
      // Increased scroll distance effect by using a larger divisor in layout
      const scrollDist = containerRect.height - windowHeight;
      let rawProgress = 0;

      if (containerRect.top <= 0) {
        rawProgress = Math.abs(containerRect.top) / scrollDist;
      }
      
      // Clamp target between 0 and 1
      const clampedTarget = Math.min(Math.max(rawProgress, 0), 1);
      targetProgress = clampedTarget;

      // LERP (Linear Interpolation) for momentum/smoothness
      // The factor 0.05 determines the "weight" or "smoothness" (lower = smoother/slower)
      currentProgress += (targetProgress - currentProgress) * 0.05;

      // Horizontal movement logic
      // Move from right (60vw) to left (-160vw)
      const startX = 60; 
      const endX = -160;
      const currentTranslate = startX + (currentProgress * (endX - startX));
      
      textTrackRef.current.style.transform = `translateX(${currentTranslate}vw)`;

      // Dreamy Spotlight Logic (Letter by Letter)
      const centerScreen = windowWidth / 2;
      
      letterRefs.current.forEach((letter) => {
        if (!letter) return;
        const rect = letter.getBoundingClientRect();
        
        // Skip calculation for off-screen letters to optimize performance
        if (rect.right < 0 || rect.left > windowWidth) return;

        const letterCenter = rect.left + (rect.width / 2);
        const dist = Math.abs(centerScreen - letterCenter);
        const maxDist = windowWidth * 0.5; // Spread of the effect
        
        // Calculate intensity (0 = edge, 1 = center)
        let intensity = 1 - (dist / maxDist);
        intensity = Math.max(0, intensity); 
        
        // Apply Dreamy Effects
        // 1. Color: Dark Grey -> White
        const lightness = 10 + (intensity * 90); // 10% to 100%
        
        // 2. Blur: 10px -> 0px
        const blur = (1 - intensity) * 10;
        
        // 3. Opacity: 0.2 -> 1
        const opacity = 0.2 + (intensity * 0.8);

        letter.style.color = `hsl(0, 0%, ${lightness}%)`;
        letter.style.filter = `blur(${blur}px)`;
        letter.style.opacity = opacity.toString();
        // Optional: slight scale effect
        letter.style.transform = `scale(${0.9 + (intensity * 0.1)})`;
      });

      animationFrameId = requestAnimationFrame(animate);
    };

    animationFrameId = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(animationFrameId);
  }, []);

  // Reset letter index for ref assignment
  let letterIndex = 0;

  return (
    <div ref={containerRef} className="relative h-[600vh] bg-black">
      <div className="sticky top-0 h-screen flex items-center overflow-hidden">
        {/* Track */}
        <div 
          ref={textTrackRef} 
          className="flex items-center pl-[10vw] will-change-transform"
        >
          {words.map((word, i) => (
            <div key={i} className="flex items-center mr-[15vw] last:mr-0">
               {word.split('').map((char, charIndex) => {
                 const index = letterIndex++;
                 return (
                   <span 
                     key={charIndex}
                     ref={el => { letterRefs.current[index] = el }}
                     className="text-[20vw] md:text-[25vh] font-bold leading-none tracking-tighter will-change-[filter,opacity,color,transform] inline-block"
                     style={{ 
                       color: '#1a1a1a',
                       transition: 'transform 0.1s linear' // small transition for scale to prevent jitter
                     }}
                   >
                     {char}
                   </span>
                 );
               })}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
};

### /components/FooterCTA.tsx
import React, { useEffect, useRef } from 'react';
import { ArrowRight } from 'lucide-react';

export const FooterCTA: React.FC = () => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = canvas.width = window.innerWidth;
    let height = canvas.height = window.innerHeight;
    let mouseX = 0;
    let mouseY = 0;

    // Particles configuration
    const particles: {x: number, y: number, speed: number, size: number, opacity: number}[] = [];
    const particleCount = 120; 

    // Initialize particles
    for (let i = 0; i < particleCount; i++) {
      particles.push({
        x: Math.random() * width,
        y: Math.random() * height,
        speed: Math.random() * 0.2 + 0.1, // Slow rising speed
        size: Math.random() * 1.5 + 0.5,
        opacity: Math.random() * 0.4 + 0.1
      });
    }

    const handleResize = () => {
       width = canvas.width = window.innerWidth;
       height = canvas.height = window.innerHeight;
    };
    
    const handleMouseMove = (e: MouseEvent) => {
        if (containerRef.current) {
            const rect = containerRef.current.getBoundingClientRect();
            mouseX = e.clientX - rect.left;
            mouseY = e.clientY - rect.top;
        }
    };

    window.addEventListener('resize', handleResize);
    window.addEventListener('mousemove', handleMouseMove);

    let animationFrameId: number;

    const animate = () => {
      ctx.clearRect(0, 0, width, height);
      
      particles.forEach(p => {
        // Move upwards
        p.y -= p.speed;
        
        // Reset when off top screen
        if (p.y < -10) {
          p.y = height + 10;
          p.x = Math.random() * width;
        }

        // Mouse interaction logic
        const dx = mouseX - p.x;
        const dy = mouseY - p.y;
        const dist = Math.sqrt(dx*dx + dy*dy);
        let alpha = p.opacity;
        let size = p.size;

        // Effect: Particles near mouse become brighter and slightly larger
        if (dist < 200) {
            const influence = (1 - dist/200);
            alpha = Math.min(0.8, p.opacity + influence * 0.5);
            size = p.size + (influence * 1);
        }

        ctx.beginPath();
        ctx.arc(p.x, p.y, size, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
        ctx.fill();
      });

      animationFrameId = requestAnimationFrame(animate);
    };

    animate();

    return () => {
      window.removeEventListener('resize', handleResize);
      window.removeEventListener('mousemove', handleMouseMove);
      cancelAnimationFrame(animationFrameId);
    };
  }, []);

  return (
    <section ref={containerRef} className="relative h-[80vh] min-h-[600px] bg-black flex items-center justify-center overflow-hidden">
      <canvas 
        ref={canvasRef} 
        className="absolute inset-0 pointer-events-none"
      />
      
      {/* Subtle vignettes for depth */}
      <div className="absolute inset-0 bg-[radial-gradient(circle_at_center,transparent_0%,rgba(0,0,0,0.8)_100%)] pointer-events-none"></div>
      
      <div className="relative z-10 flex flex-col items-center text-center px-6">
        <h2 className="text-5xl md:text-8xl lg:text-[9rem] font-bold text-white tracking-tighter mb-6 leading-none select-none drop-shadow-2xl">
          QYVORA
        </h2>
        
        <p className="text-white/60 max-w-md text-lg md:text-xl font-light mb-12 tracking-wider uppercase text-sm">
          The clarity layer for climate action
        </p>

        <button className="group relative px-12 py-5 bg-white text-black rounded-sm overflow-hidden transition-all duration-500 hover:shadow-[0_0_40px_rgba(255,255,255,0.3)]">
           <span className="relative z-10 font-medium tracking-widest text-xs md:text-sm flex items-center gap-3 uppercase">
             Get climate updates <ArrowRight size={16} className="group-hover:translate-x-1 transition-transform" />
           </span>
           <div className="absolute inset-0 bg-neutral-200 translate-y-full group-hover:translate-y-0 transition-transform duration-300 ease-out"></div>
        </button>
      </div>
    </section>
  );
};
