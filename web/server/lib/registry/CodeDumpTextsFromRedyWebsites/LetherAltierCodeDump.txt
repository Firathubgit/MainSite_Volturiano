# Codebase Overview & Rendering Order

## Application Entry Point
1.  **`index.tsx`**: This is the entry point. It finds the root element and renders the `App` component wrapped in `React.StrictMode`.

## Component Rendering Hierarchy (App.tsx)
The `App` component acts as the main layout container. Components are rendered in this specific vertical order:

1.  **`Navbar`** (`components/Navbar.tsx`):
    - Rendered at the top, outside the `<main>` tag.
    - It is `fixed` positioned and handles its own visibility based on scroll.

2.  **`main`** Content Area:
    - **`Hero`** (`components/Hero.tsx`):
        - The first section visible.
        - Contains the video background, the main "RIVELON" title, and the "Explore" button.
        - Uses **`Button`** (`components/Button.tsx`).
    
    - **`ProductShowcase`** (`components/ProductShowcase.tsx`):
        - A full-screen section showcasing a specific product (The Drifter Boot).
        - Contains the product image and details.

    - **`FeatureScroll`** (`components/FeatureScroll.tsx`):
        - A logical "scrollytelling" section where users scroll through product features (Goodyear welt, Vibram sole, etc.).
        - It uses a sticky layout to keep the text/image in view while content changes.

    - **`CommunityGrid`** (`components/CommunityGrid.tsx`):
        - A grid of images ("Worn in the Wild") showing the community/lifestyle shots.

    - **`KeepExploring`** (`components/KeepExploring.tsx`):
        - A navigation section with cards leading to other parts of the site (Store, Accessories, etc.).

    - **`Footer`** (Inline in `App.tsx`):
        - The bottom section with contact links, social links, and the branding stamp.

## Available Components (Not currently rendered in App.tsx)
- **`Philosophy.tsx`**: A purely presentational section with a "Manifesto" and image. It is available in the components folder but not currently used in the main flow of `App.tsx`.
- **`Concierge.tsx`**: An AI-powered chat interface component. It is fully implemented with `geminiService.ts` but not currently placed in the render tree of `App.tsx`.

## Utilities & Services
- **`types.ts`**: TypeScript definitions for shared interfaces like `ChatMessage` and `SectionId`.
- **`services/geminiService.ts`**: Handles the communication with the Google Gemini API for the Concierge component.
- **`index.css`**: Global styles, Tailwind directives, and the specific CSS for the "Volturiano Stamp".

```tsx
import React, { useState, useEffect } from 'react';
import { SectionId } from '../types';

const Navbar: React.FC = () => {
  const [isVisible, setIsVisible] = useState(true);
  const [isAtTop, setIsAtTop] = useState(true);

  useEffect(() => {
    const handleScroll = () => {
      const currentScrollY = window.scrollY;

      // Only visible at the very top (threshold < 50px)
      const atTop = currentScrollY < 50;
      setIsVisible(atTop);
      setIsAtTop(atTop);
    };

    window.addEventListener('scroll', handleScroll, { passive: true });

    handleScroll();
```

```tsx
const FeatureScroll: React.FC = () => {
  const sectionRef = useRef<HTMLDivElement>(null);
  const [activeIndex, setActiveIndex] = useState(0);

  useEffect(() => {
    const handleScroll = () => {
      if (!sectionRef.current) return;

      const { top, height } = sectionRef.current.getBoundingClientRect();
      const viewportHeight = window.innerHeight;
      const totalScrollDistance = height - viewportHeight;

      // Calculate how far we've scrolled into the section (0 to 1)
      let progress = -top / totalScrollDistance;
      progress = Math.max(0, Math.min(1, progress));

      // Determine active index
      const newIndex = Math.min(features.length - 1, Math.floor(progress * features.length));
      setActiveIndex(newIndex);
    };

    window.addEventListener('scroll', handleScroll, { passive: true });
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  return (
    // Reduced height from 400vh to 250vh to effectively halve the scroll effort
    <section ref={sectionRef} className="h-[250vh] bg-black relative">

      <div className="sticky top-0 h-screen w-full flex flex-col md:flex-row p-4 md:p-24 gap-4 md:gap-24 overflow-hidden box-border">

        {/* Left Card: Text List (Top on mobile) */}
        <div className="w-full h-[45vh] md:h-auto md:flex-1 bg-white rounded-[1.5rem] md:rounded-[2rem] relative overflow-hidden shadow-2xl order-1">

          <div className="absolute top-6 left-6 md:top-12 md:left-20 z-10">
            <span className="text-[10px] md:text-xs font-mono uppercase tracking-widest text-stone-400">Details</span>
          </div>

          {/* Text Container: Positioned absolute top-1/2, then translated to center the ACTIVE item */}
          <div
            className="absolute top-1/2 left-0 w-full px-6 md:px-20 transition-transform duration-500 ease-out will-change-transform"
            style={{ transform: `translateY(calc(-${activeIndex * 140}px - 70px))` }}
          >
            {features.map((feature, i) => (
              <div
                key={i}
                className={`h-[140px] flex flex-col justify-center transition-all duration-500 ${i === activeIndex ? 'opacity-100 blur-0 translate-x-0' : 'opacity-20 blur-[2px] translate-x-4'}`}
              >
                <h2 className="text-3xl md:text-6xl font-sans font-medium tracking-tight text-neutral-900 leading-none">
                  {feature.title}
                </h2>
                {/* Only show description for active item */}
                <div className={`overflow-hidden transition-all duration-500 ${i === activeIndex ? 'max-h-24 opacity-100 mt-2 md:mt-4' : 'max-h-0 opacity-0 mt-0'}`}>
                  <p className="text-neutral-600 text-xs md:text-base max-w-md leading-relaxed line-clamp-2 md:line-clamp-none">
                    {feature.description}
                  </p>
                </div>
              </div>
            ))}
          </div>

          {/* Progress Indicator Line */}
          <div className="absolute left-6 md:left-12 top-1/2 -translate-y-1/2 w-0.5 h-32 md:h-64 bg-neutral-200 rounded-full overflow-hidden">
            <div
              className="w-full bg-neutral-900 transition-all duration-300"
              style={{ height: `${(activeIndex + 1) / features.length * 100}%` }}
            ></div>
          </div>
        </div>

        {/* Right Card: Images (Bottom on mobile) */}
        <div className="w-full h-[45vh] md:h-auto md:flex-1 bg-[#292524] rounded-[1.5rem] md:rounded-[2rem] relative overflow-hidden shadow-2xl order-2">
```

## /services/geminiService.ts
```ts
import { GoogleGenAI } from "@google/genai";

const apiKey = process.env.API_KEY || '';
const ai = new GoogleGenAI({ apiKey });

export const sendConciergeMessage = async (
  history: { role: 'user' | 'model'; text: string }[],
  newMessage: string
): Promise<string> => {
  try {
    const model = 'gemini-3-flash-preview';
    
    // Construct the conversation history for the model
    const recentHistory = history.slice(-6).map(msg => ({
      role: msg.role,
      parts: [{ text: msg.text }]
    }));

    const chat = ai.chats.create({
      model,
      history: recentHistory,
      config: {
        systemInstruction: `You are 'The Rancher', a digital concierge for 'Aura', a brand blending high-end web design with the spirit of the American West.
        
        Your persona:
        - Rugged, sophisticated, and calm.
        - You use metaphors related to the desert, horses, stars, and the open range.
        - You value silence and only speak when it matters ("Less is more").
        - Your tone is warm but firm, like worn leather.
        - Do NOT talk like a cartoon cowboy (avoid excessive "Yeehaw"). Think more "Yellowstone" or Cormac McCarthy.
        
        If asked about the site: It is a digital sanctuary for those who seek clarity in the noise.
        If asked about the video: "Time flows like a river through the canyon. We just watch it pass."
        
        Keep responses concise (under 3 sentences).`,
        temperature: 0.7,
        maxOutputTokens: 150,
      }
    });

    const result = await chat.sendMessage({ message: newMessage });
    return result.text || "The wind is howling. I didn't quite catch that.";
  } catch (error) {
    console.error("Gemini API Error:", error);
    return "The line is down. Storm's interfering with the signal.";
  }
};
```
